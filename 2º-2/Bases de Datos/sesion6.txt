select id_pedido, count(*) as numero_productos
from detalles_pedido
group by id_pedido;

select id_pedido, count(*) as numero_productos,
	sum(cantidad) as total_usuarios,
	sum(cantidad*precio) as impore_total
from detalles_pedido inner join productos using(id_producto)
group by id_pedido
order by impore_total desc limit 4;

select id_cliente, sum(cantidad*precio) as gasto_total
from pedidos inner join detalles_pedido using(id_pedido)
	inner join productos using(id_producto)
where id_cliente is not null
group by id_cliente;

select id_cliente, c.nombre, sum(cantidad*precio) as gasto_total,
	round(avg(cantidad*precio), 3) as media_producto
from clientes c inner join pedidos using(id_cliente)
	inner join detalles_pedido using(id_pedido)
	inner join productos using(id_producto)
group by id_cliente;

select id_producto, nombre, existencias,
	sum(cantidad) as numero_unidades_vendidas,
	existencias - sum(cantidad) as existencias_actual
from detalles_pedido inner join productos using(id_producto)
group by id_producto;

select id_producto, nombre, existencias,
	sum(cantidad) as numero_unidades_vendidas,
	existencias - sum(cantidad) as existencias_actual
from productos inner join detalles_pedido using(id_producto)
group by id_producto
order by existencias_actual desc, numero_unidades_vendidas desc;
