create or replace procedure getEstudianteMatriculado(idAlumno integer, nombreAsignatura varchar)
as $$
declare
	nombre_completo text;
	existe integer;
begin
select concat(' ', nombre, apellido1, apellido2)
    into nombre_completo
    from alumno
    where id_alumno = idAlumno;

    select count(*)
    into existe
    from matricula m
    inner join asignatura a on m.id_asignatura = a.id_asignatura
    where m.id_alumno = idAlumno
      and lower(a.nombre) = lower(nombreAsignatura);

    if existe > 0 then
        raise notice 'Alumno % matriculado en %', nombre_completo, nombreAsignatura;
    else
        raise notice 'Alumno % NO matriculado en %', idAlumno, nombreAsignatura;
    end if;
end;
$$ language plpgsql;

call getEstudianteMatriculado(19, 'Sistemas distribuidos');
call getEstudianteMatriculado(2, 'Seguridad');



create or replace function geTiposAsignaturasCiudad()
returns table (tipo varchar(20), ciudad varchar(20), cantidad bigint)
as $$
begin
    return query
    select a.tipo, p.ciudad, count(*) as cantidad
    from asignatura a
    inner join profesor p on a.id_profesor = p.id_profesor
    where lower(p.ciudad) in ('gijon', 'oviedo')
    group by a.tipo, p.ciudad
    having count(*) > 2;
end;
$$ language plpgsql;

select * from geTiposAsignaturasCiudad();


create or replace function getProfesoresSalarioMayorMedia(p_departamento varchar)
returns table (id_profesor integer, nombre varchar(20), apellido1 varchar(20), salario decimal(10,2))
as $$
begin
    return query
    select p.id_profesor, p.nombre, p.apellido1, p.salario
    from profesor p
    inner join departamento d on p.id_dpto = d.id_dpto
    where lower(d.nombre) = lower(p_departamento)
      and p.salario > (select avg(p2.salario) from profesor p2);
end;
$$ language plpgsql;

create or replace function infoProfesores()
returns trigger
as $$
begin
    if tg_op = 'INSERT' then
        raise notice 'El profesor % % % ha sido insertado', new.nombre, new.apellido1, new.apellido2;
        return new;

    elsif tg_op = 'UPDATE' then
        raise notice 'Se ha actualizado la informacion del profesor %', new.id_profesor;

        if old.salario is distinct from new.salario then
            raise notice 'El salario del profesor % cambia de % a % euros', new.id_profesor, old.salario, new.salario;
        end if;

        return new;

    elsif tg_op = 'DELETE' then
        raise notice 'El profesor % % % ha sido eliminado', old.nombre, old.apellido1, old.apellido2;
        return old;
    end if;

    return null;
end;
$$ language plpgsql;


create trigger tg_infoProfesores
after insert or update or delete on profesor
for each row execute function infoProfesores();

create table historialMatriculas (
    id_alumno integer NOT NULL,
    id_asignatura integer NOT NULL,
    calificacion varchar(20) DEFAULT 'Aprobado',
    fecha timestamp,
    PRIMARY KEY(id_alumno, id_asignatura),
    FOREIGN KEY(id_alumno) references alumno(id_alumno),
    FOREIGN KEY(id_asignatura) references asignatura(id_asignatura)
);


create or replace function historialMatriculas()
returns trigger
as $$
begin
    insert into historialMatriculas(id_alumno, id_asignatura, fecha)
    values (old.id_alumno, old.id_asignatura, now());

    raise notice 'Registro almacenado en el historial de matriculas';

    return old;
end;
$$ language plpgsql;

create trigger tg_historialMatriculas
after delete on matricula
for each row execute procedure historialMatriculas();
