-- ============================================================
-- EJERCICIO 1
-- Funcion perfiles_en_red(red text)
--
-- Idea:
--   La tabla perfil relaciona cada persona con una red social.
--   Por eso se juntan persona, perfil y red_social.
--   La funcion devuelve una tabla con los datos pedidos.
--   Si la consulta no devuelve filas, se muestra un NOTICE.
-- ============================================================

DROP FUNCTION IF EXISTS perfiles_en_red(text);

CREATE OR REPLACE FUNCTION perfiles_en_red(red text)
RETURNS TABLE (
    dni VARCHAR(9),
    nombre VARCHAR(40),
    apellido VARCHAR(40),
    usuario VARCHAR(40),
    tipo_cuenta VARCHAR(20),
    nombre_red VARCHAR(40)
)
AS $$
BEGIN
    RETURN QUERY
    SELECT  pe.dni,
            pe.nombre,
            pe.apellido,
            pf.usuario,
            pf.tipo_cuenta,
            rs.nombre
    FROM persona pe
         INNER JOIN perfil pf
             ON pe.dni = pf.dni_persona
         INNER JOIN red_social rs
             ON pf.nombre_red = rs.nombre
    WHERE rs.nombre = red
    ORDER BY pe.apellido, pe.nombre;

    IF NOT FOUND THEN
        RAISE NOTICE 'No hay personas con perfil en la red social %', red;
    END IF;
END;
$$ LANGUAGE plpgsql;

-- Prueba con resultados:
SELECT * FROM perfiles_en_red('Instagram');

-- Prueba sin resultados:
SELECT * FROM perfiles_en_red('Twitter');


-- ============================================================
-- EJERCICIO 2
-- Funcion listar_relaciones_persona(persona_dni text)
--
-- Nota:
--   Uso el nombre sin tilde para evitar problemas con identificadores
--   SQL. Si el profesor exige exactamente el nombre con tilde, bastaria
--   con crear la funcion entre comillas con ese nombre.
--
-- Idea:
--   Se buscan las relaciones donde la persona dada es el origen.
--   Para cada fila se muestra un mensaje con RAISE NOTICE.
--   El recorrido se hace con FOR ... IN SELECT, que es el bucle
--   especifico de PL/pgSQL para recorrer filas sin cursor explicito.
-- ============================================================

DROP FUNCTION IF EXISTS listar_relaciones_persona(text);

CREATE OR REPLACE FUNCTION listar_relaciones_persona(persona_dni text)
RETURNS void
AS $$
DECLARE
    fila RECORD;
    contador INTEGER := 0;
BEGIN
    FOR fila IN
        SELECT  po.nombre AS nombre_origen,
                po.apellido AS apellido_origen,
                pd.nombre AS nombre_destino,
                pd.apellido AS apellido_destino,
                r.tipo_relacion,
                r.detalle
        FROM relacion r
             INNER JOIN persona po
                 ON r.dni_origen = po.dni
             INNER JOIN persona pd
                 ON r.dni_destino = pd.dni
        WHERE r.dni_origen = persona_dni
        ORDER BY pd.apellido, pd.nombre, r.tipo_relacion
    LOOP
        contador := contador + 1;

        RAISE NOTICE '% % tiene relacion con % %. Tipo: %. Detalle: %',
            fila.nombre_origen,
            fila.apellido_origen,
            fila.nombre_destino,
            fila.apellido_destino,
            fila.tipo_relacion,
            fila.detalle;
    END LOOP;

    IF contador = 0 THEN
        RAISE NOTICE 'No hay relaciones registradas para la persona con DNI %', persona_dni;
    END IF;
END;
$$ LANGUAGE plpgsql;

-- Prueba con resultados:
SELECT listar_relaciones_persona('11111111A');

-- Prueba sin resultados, porque Diego no aparece como origen en los datos iniciales:
SELECT listar_relaciones_persona('66666666F');


-- ============================================================
-- EJERCICIO 3
-- Funcion mostrar_redes_con_num_personas()
--
-- Idea:
--   Se declara un cursor explicito sobre todas las redes sociales.
--   Para cada red recuperada con FETCH, se calcula el numero de
--   personas que tienen perfil en esa red.
--   COUNT(DISTINCT dni_persona) cuenta personas, no perfiles repetidos.
-- ============================================================

DROP FUNCTION IF EXISTS mostrar_redes_con_num_personas();

CREATE OR REPLACE FUNCTION mostrar_redes_con_num_personas()
RETURNS void
AS $$
DECLARE
    cur_redes CURSOR FOR
        SELECT nombre, empresa, fecha_lanzamiento
        FROM red_social
        ORDER BY nombre;

    red_actual RECORD;
    num_personas INTEGER;
BEGIN
    OPEN cur_redes;

    LOOP
        FETCH cur_redes INTO red_actual;
        EXIT WHEN NOT FOUND;

        SELECT COUNT(DISTINCT dni_persona)
        INTO num_personas
        FROM perfil
        WHERE nombre_red = red_actual.nombre;

        RAISE NOTICE 'Red: %. Empresa: %. Fecha de lanzamiento: %. Numero de personas: %',
            red_actual.nombre,
            red_actual.empresa,
            red_actual.fecha_lanzamiento,
            num_personas;
    END LOOP;

    CLOSE cur_redes;
END;
$$ LANGUAGE plpgsql;

-- Prueba:
SELECT mostrar_redes_con_num_personas();


-- ============================================================
-- EJERCICIO 4.a
-- Tabla de log de operaciones sobre relacion
--
-- Idea:
--   La tabla guarda:
--      - un identificador del log,
--      - el momento de la operacion,
--      - el tipo de operacion: INSERT, UPDATE o DELETE,
--      - todos los campos de la fila afectada en relacion.
--
--   Se eliminan antes trigger, funcion y tabla para poder volver a
--   ejecutar este script sin errores.
-- ============================================================

DROP TRIGGER IF EXISTS trg_log_relacion ON relacion;
DROP FUNCTION IF EXISTS registrar_operacion_relacion();
DROP TABLE IF EXISTS log_operaciones_relacion;

CREATE TABLE log_operaciones_relacion (
    id_log SERIAL PRIMARY KEY,
    momento_operacion TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    tipo_operacion VARCHAR(10) NOT NULL,
    dni_origen VARCHAR(9),
    dni_destino VARCHAR(9),
    tipo_relacion VARCHAR(20),
    detalle VARCHAR(80)
);


-- ============================================================
-- EJERCICIO 4.b
-- Funcion disparadora sobre la tabla relacion
--
-- Idea:
--   TG_OP indica la operacion que ha activado el trigger.
--   En INSERT y UPDATE se usan los datos de NEW.
--   En DELETE se usan los datos de OLD, porque la fila ya no existira
--   despues de borrar.
-- ============================================================

CREATE OR REPLACE FUNCTION registrar_operacion_relacion()
RETURNS trigger
AS $$
BEGIN
    IF TG_OP = 'INSERT' THEN
        INSERT INTO log_operaciones_relacion (
            momento_operacion,
            tipo_operacion,
            dni_origen,
            dni_destino,
            tipo_relacion,
            detalle
        )
        VALUES (
            CURRENT_TIMESTAMP,
            TG_OP,
            NEW.dni_origen,
            NEW.dni_destino,
            NEW.tipo_relacion,
            NEW.detalle
        );

        RETURN NEW;

    ELSIF TG_OP = 'UPDATE' THEN
        INSERT INTO log_operaciones_relacion (
            momento_operacion,
            tipo_operacion,
            dni_origen,
            dni_destino,
            tipo_relacion,
            detalle
        )
        VALUES (
            CURRENT_TIMESTAMP,
            TG_OP,
            NEW.dni_origen,
            NEW.dni_destino,
            NEW.tipo_relacion,
            NEW.detalle
        );

        RETURN NEW;

    ELSIF TG_OP = 'DELETE' THEN
        INSERT INTO log_operaciones_relacion (
            momento_operacion,
            tipo_operacion,
            dni_origen,
            dni_destino,
            tipo_relacion,
            detalle
        )
        VALUES (
            CURRENT_TIMESTAMP,
            TG_OP,
            OLD.dni_origen,
            OLD.dni_destino,
            OLD.tipo_relacion,
            OLD.detalle
        );

        RETURN OLD;
    END IF;

    RETURN NULL;
END;
$$ LANGUAGE plpgsql;


-- ============================================================
-- EJERCICIO 4.c
-- Vincular la funcion con INSERT, UPDATE y DELETE sobre relacion
--
-- Se usa FOR EACH ROW porque queremos guardar una entrada de log por
-- cada fila insertada, modificada o borrada.
-- ============================================================

CREATE TRIGGER trg_log_relacion
AFTER INSERT OR UPDATE OR DELETE
ON relacion
FOR EACH ROW
EXECUTE PROCEDURE registrar_operacion_relacion();
