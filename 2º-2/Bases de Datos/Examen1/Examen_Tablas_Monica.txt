create table nave_espacial(
	id_nave_espacial integer,
	nombre_nave_espacial varchar(30),
	max_numero_tripulante integer,
	primary key(id_nave_espacial)
);

create table tripulante(
	id_tripulante integer,
	id_nave_espacial integer,
	nombre_tripulante varchar(20),
	apellidos_tripulante varchar(20),
	pais_tripulante varchar(20),
	primary key(id_tripulante)
);
