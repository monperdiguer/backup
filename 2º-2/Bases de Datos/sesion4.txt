select es.id_enfermedad, nombre, count(*) as
numeropacientes
from enfermedades en inner  join estancias es
on en.id_enfermedad = es.id_enfermedad
where nombre = 'Gripe'
union
select es.id_enfermedad, nombre, count(*) as
numeropacientes
from enfermedadesen inner join estancias es
on en.id_enfermedad = es.id_enfermedad
where nomre = 'Ulcera';

select d.id_doctor, d.nombre
from especialidades_doctores ed inner join
doctores d
on ed.id_especialidad = d.id_especialidad
where ed.nombre = 'Cardiologia'

intersect

select d.id_doctor, d.nombre
from doctores d inner join dirigir_hospitales dh
on d.id_doctor = dh.id_doctor;

select nombre, count(*) as numero_doctores
from hospitales h, trabajar_doctores td
where h.id_hospital = td.id_hospital
group by nombre
having nombre = 'La Paz'

union

select nombre, count(*) as numero_doctores
from hospitales h, trabajar_doctores td
where h.id_hospital = td.id_hospital
group by nombre
having nombre = 'General';

select d.id_doctor, d.nombre, h.nombre
from especialidades_doctores ed,
	 doctores d,
	 trabajar_doctores td,
	 hospitales h
where ed.id_especialidad = d.id_especialidad
	   and
	   d.id_doctor = td.id_doctor
	   and
	   td.id_hospital = h.id_hospital
	   and
	   ed.nombre = 'Pediatria'

intersect

select d.id_doctor, d.nombre, h.nombre
from doctores d, dirigir_hospitales dh,
	 hospitales h
where d.id_doctor = dh.id_doctor and
	 dh.id_hospital = h.id_hospital;
